\begin{frame}
  \frametitle{構文解析(Parsing)}

  文字列のならびが規則通りにならんでいるかチェックすること。

  \vspace*{.7cm}

  \begin{tabular}{lcl}
    字句解析                    & ： & 正規表現(Regular Expression)           \\[10pt]
    \greentext{(トークンにまとめる)} &   & $FA$ : 有限オートマトン                    \\[5pt]
                            &   & $w\stackrel{\mbox{?}}{\in} L(FA)$  \\[10pt]
    構文解析                    & ： & 文脈自由文法(Context Free Language)      \\[10pt]
    \greentext{(木構造を構成)}    &   & $PDA$ : プッシュダウンオートマトン              \\[5pt]
                            &   & $w\stackrel{\mbox{?}}{\in} L(PDA)$ \\
  \end{tabular}

  \vspace*{.7cm}

  木構造を出力：$PDT$ \greentext{プッシュダウントランスデューサー}

\end{frame}

\begin{frame}
  \frametitle{構文解析の基礎}
  \begin{list}{}{}
    \item [\bluetext{{\bf 構文解析}}]\ \\
          字句解析の出力トークン列から文法に従った木構造を構成\\[5pt]
    \item [\bluetext{{\bf 文法}}] 文脈自由文法(CF文
          法)\quad$(N,\Sigma,P,E)$\\
          非終端記号:$N=\{E\}$,終端記号:$\Sigma=\{+,*,(,)\}\cup\mathtt{ID}$
          \[
            P=
            \left\{
            \begin{array}{ll}
              E\rightarrow E+E,        &
              E\rightarrow E*E,                              \\
              E\rightarrow\mathtt{ID}, & E\rightarrow(\ E\ )
            \end{array}
            \right\}
          \]
          \begin{list}{(\arabic{coffee})}{\usecounter{coffee}\setlength{\itemsep}{5pt}}
            \item ${\mathtt{ID}}$は式である。
            \item $x$と$y$が式であれば、$x+y$,$x*y$,$(x)$は式である。
            \item (1),(2)を有限回適用して得られる表現だけが式である。
          \end{list}
    \item {\bf BNF(Backus-Nauer-Form)}
          \[
            \tuple{S}::=\tuple{S}+\tuple{S}\ |\ \tuple{S}*\tuple{S}\ |
            \ {\mathtt{ID}}\ |\ (\tuple{S})
          \]
  \end{list}
\end{frame}

\begin{frame}
  \frametitle{導出と還元}
  \begin{list}{}{\setlength{\itemsep}{5pt}}
    \item [\bluetext{{\bf 導出}}]：$w_1\Rightarrow w_2$\\
          $w_1,w_2\in (N\cup\Sigma)^\ast$
          $w_1=wAw''$, $w_2=wvw'$ に対して $A\rightarrow v$
    \item [\bluetext{{\bf 導出関係}}]\ \\
          $v\stackrel{\ast}{\Rightarrow} v'$:$\Rightarrow$の0回以上の適
          用
          \[
            v=v_0\Rightarrow v_1\Rightarrow\cdots\Rightarrow
            v_{n-1}\Rightarrow v_n=v'
          \]
    \item [\bluetext{\bf 還元}] $w_1\Rightarrow w_2$\\
          $w_2$は$w_1$に還元される。
    \item [\bluetext{\bf 生成言語}]\ \\
          $G=\tuple{N,\Sigma,P,S}$が生成する言語：$L(G)=\{w\ |\
            S\stackrel{\ast}{\Rightarrow}w\}\subseteq \Sigma^\ast$
  \end{list}
  \greentext{\small
    $w\in L(G)$のとき、文法に従った文、$w\notin L(G)$のとき、文法エラー}\\
  構文解析=$w\in L(G)$をチェックし、$w\in L(G)$のとき解析木を生成
\end{frame}

\begin{frame}
  \frametitle{最左導出、最右導出}
  $A\rightarrow\alpha\in P$によって、$\beta A\gamma$を書き換える。
  \[
    \beta A\gamma\Rightarrow \beta\alpha\gamma
  \]
  \begin{list}{}{}
    \item 最左導出：$\beta\in\Sigma^\ast$ \quad $\Rightarrow_{lm}$
    \item 最右導出：$\gamma\in\Sigma^\ast$ \quad $\Rightarrow_{rm}$
  \end{list}
  \only<1>{
    \begin{eqnarray*}
      \mbox{関数定義} & \Rightarrow_{lm} & \mbox{型名}\ \mbox{名前}\
      '\mathtt{(}'\ '\mathtt{)}'\ \mbox{複文}\\
      & \Rightarrow_{lm} & \mathtt{INT}\ \mbox{名前}\
      '\mathtt{(}'\ '\mathtt{)}'\ \mbox{複文}\\
      & \Rightarrow_{lm} & \mathtt{INT}\ \mathtt{NAME}\
      '\mathtt{(}'\ '\mathtt{)}'\ \mbox{複文}\\
      & \Rightarrow_{lm} & \mathtt{INT}\ \mathtt{NAME}\
      '\mathtt{(}'\ '\mathtt{)}'\ \mathtt{COMPOUND}\\
    \end{eqnarray*}
  }
  \only<2>{
    \begin{eqnarray*}
      \mbox{関数定義} & \Rightarrow_{rm} & \mbox{型名}\ \mbox{名前}\
      '\mathtt{(}'\ '\mathtt{)}'\ \mbox{複文}\\
      & \Rightarrow_{rm} & \mbox{型名}\ \mbox{名前}\
      '\mathtt{(}'\ '\mathtt{)}'\ \mathtt{COMPOUND}\\
      & \Rightarrow_{rm} & \mbox{型名}\ \mathtt{NAME}\
      '\mathtt{(}'\ '\mathtt{)}'\ \mathtt{COMPOUND}\\
      & \Rightarrow_{rm} & \mathtt{INT}\ \mathtt{NAME}\
      '\mathtt{(}'\ '\mathtt{)}'\ \mathtt{COMPOUND}\\
    \end{eqnarray*}
  }
\end{frame}

\begin{frame}{First/Followの適用}
  同じ左辺を持つ複数の生成規則がある場合

  \[\begin{array}{crcl}
      1: & A & \rightarrow & \alpha \\
      2: & A & \rightarrow & \beta
    \end{array}\]

  \greentext{
    $\varepsilon\notin First(\alpha)\wedge\varepsilon\notin First(\beta)$の場合
  }

  最左導出の先頭の文字がFirst($\alpha$)である場合，規則１を適用\\
  最左導出の先頭の文字がFirst($\beta$)である場合，規則２を適用\\

  \vspace*{10pt}

  \greentext{
  $\varepsilon\in First(\alpha)\vee\varepsilon\in First(\beta)$の場合
  }

  ($\varepsilon\in First(\alpha)$とすると)
  最左導出の先頭の文字が$First(\alpha)\backslash\{\varepsilon\}\cup Follow(\alpha)$の場合，規則１を適用\\
  最左導出の先頭の文字がFirst($\beta$)である場合，規則２を適用

  \vspace*{10pt}

  \redtext{上記の方法で適用する規則が決まれば決定的に最左導出が可能}

\end{frame}

\begin{frame}
  \frametitle{再帰降下法の問題点}
  \begin{list}{}{}
    \item 生成規則の左辺に同じ非終端記号をもつ規則が複数ある場合、どの規則
          を適用すべきか？\\
          \greentext{まちがった規則を適用すると解析木が生成できない}

          \[
            \{A\rightarrow a,A\rightarrow b\}
          \]
          に対して、入力$b$に対して、1番目の規則を適用しても構文解析できな
          い。\greentext{後戻り(back track)して別の規則を適用すればよい}

          後戻りは計算機として一般に重い処理\\
          \quad$\rightarrow$あらかじめどの規則を適用できるか知りたい\\
          \qquad 生成規則を予め解析して$First/Follow$を得る\\
          \quad 入力の文字を先読みする(Look-ahead)

          先の例は、次の文字が$b$であることを知っていたら、2番目の規則を後
          戻りなく適用可能
  \end{list}
\end{frame}

\begin{frame}

  生成規則に以下のような規則が含まれるとする。

      A & \rightarrow & \alpha\beta_2 \\
    \end{array}
  \]

  \[

      A  & \rightarrow & \alpha A' \\
      A' & \rightarrow & \beta_1   \\
      A' & \rightarrow & \beta_2   \\
    \end{array}
  \]
  \redtext{導出は変わらない}

  降下型構文解析に不要なバックトラックを避ける。

  \greentext{そのままでは下向き構文解析不能であっても、等価で解析可能な文
    法に変換できる}
\end{frame}

\begin{frame}
  \frametitle{左再帰性}

  \begin{itemize}
    \item 右辺の非終端記号の呼出の順序はなんでもよい。
          生成規則の右辺に左辺と同じ記号が表れる。(再帰性)\\
          最初から処理していく場合に、\\
          右辺の最初に同じ非終端記号あらわれたら？(左再帰性)
  \end{itemize}

  {\scriptsize
  \begin{eqnarray*}
    \mbox{パラメータリスト} & \rightarrow & \mbox{パラメータ}\
    \mbox{パラメータリスト}\\
    \mbox{パラメータリスト} & \rightarrow & \mbox{パラメータ}
  \end{eqnarray*}
  }

  ずっと最初の規則を適用することがおこりうる。\\

  パラメータリスト$\not\Rightarrow^\ast\varepsilon$であるとすると、\\

  First(パラメータリスト　パラメータ)=First(パラメータリスト)\\

  {\color[rgb]{0,0.5,0}２つの規則はFirstで見分けがつかない}
\end{frame}
\begin{frame}
  \frametitle{左再帰性除去}
  左再帰性は以下のように除去可能

  {\scriptsize
  \begin{eqnarray*}
    \mbox{パラメータリスト} & \rightarrow & \mbox{パラメータ}\
    \mbox{パラメータリスト'}\\
    \mbox{パラメータリスト'} & \rightarrow & \mbox{','}\ \mbox{パラメータ}\
    \mbox{パラメータリスト'}\\
    \mbox{パラメータリスト'} & \rightarrow & \varepsilon\\
  \end{eqnarray*}
  }

  一般に
  \uncover<1->{
    \begin{center}
      $\begin{array}{l}
          A\rightarrow A\alpha_1, A\rightarrow A\alpha_2,\cdots,A\rightarrow A\alpha_m \\
          A\rightarrow \beta_1,A\rightarrow \beta_2,\cdots ,A\rightarrow \beta_n
        \end{array}$
    \end{center}
    は
  }
  \uncover<2>{
    \begin{center}
      $\begin{array}{l}
          A\rightarrow\beta_1 A',A\rightarrow\beta_2 A',\cdots,A\rightarrow\beta_n A' \\
          A'\rightarrow\alpha_1 A',A'\rightarrow\alpha_2 A',\cdots,A'\rightarrow\alpha_m A',A'\rightarrow\varepsilon
        \end{array}$
    \end{center}
    という等価な文法に変換可能$\Rightarrow$\redtext{左再帰性除去}
  }
\end{frame}

    \item $\varepsilon\in\mathsf{First}_1(\alpha_i)$ならば、$A$の行、$a\in\mathsf{Follow}_1(A)$の列に$A\rightarrow\alpha_i$を追加
  \end{itemize}

  \begin{center}{\scriptsize

        \begin{itemize}
          \item $e=\varepsilon$のとき: $\{\varepsilon\}$, $e\in\Sigma$のとき: $\{e\}$
          \item $e=e_1\vert e_2$のとき: $\mathsf{First}_1(e_1)\cup\mathsf{First}_2(e_2)$
          \item $e=e_1e_2$のとき:

                  \item $\varepsilon\notin\mathsf{First}_1(e_1)$のとき　$\mathsf{First}_1(e_1)$
                \end{list}
          \item $e=e_1^\ast$のとき: $\mathsf{First}_1(e_1)\cup\{\varepsilon\}$
        \end{itemize}
      \end{itembox}

      \begin{itembox}[l]{定義4.6 [$Follow_1(e)$]}
        \begin{itemize}
          \item $A\rightarrow e\in P$のとき $\mathsf{Follow}_1(e):=\mathsf{Follow_1}(A)$
          \item $e=e_1|e_2$のとき　$\mathsf{Follow}_1(e_1):=\mathsf{Follow}_1(e)$, $\mathsf{Follow}_1(e_2):=\mathsf{Follow}_1(e)$
          \item $e=e_1e_2$のとき　$\mathsf{Follow}_1(e_2):=\mathsf{Follow}_1(e)$\\
                \begin{list}{}{}
                  \item $\varepsilon\in\mathsf{First}_1(e_2)$のとき$\mathsf{Follow}_1(e_1):=\mathsf{First}_1(e_2)\backslash\{\varepsilon\}\cup\mathsf{Follow}_1(e)$
                  \item $\varepsilon\notin\mathsf{First}_1(e_2)$のとき$\mathsf{Follow}_1(e_1):=\mathsf{First}_1(e_2)$
                \end{list}
          \item $e=e^\ast$のとき　$\mathsf{Follow}_1(e_1):=\mathsf{First}_1(e_1)\backslash\{\varepsilon\}\cup\mathsf{Follow}_1(e)$
        \end{itemize}
      \end{itembox}

  \frametitle{jjtreeによる再帰降下型パーザの生成}

  \verb|javacc|に含まれる\verb|jjtree|をつかうと文脈自由文法から再帰降下
  構文解析プログラムを生成できる。

  \begin{center}
    \includegraphics[width=.6\textwidth]{jjtree.pdf}
  \end{center}

  \begin{list}%

    {} %formatting parameter
    \item [(1)]
          字句定義と文脈自由文法を記述したファイル(\verb|jjt|ファイル)を\verb|jjtree|で変換して、
          いくつかの\verb|java|ファイルと\verb|javacc|ファイル(\verb|jj|ファイル)を生成する。
    \item [(2)]
          \verb|javacc|のファイルを\verb|javacc|で\verb|java|に変換して、全部を
          \verb|class|ファイルに変換して完成
  \end{list}
\end{frame}

\begin{frame}[containsverbatim]
  \frametitle{例(四則演算式)}

  以下の(拡張)BNF定義で定義される文法の構文解析を考える．

  \[\begin{array}{rcl}
      \NTerm{AddExpr}  & \mathtt{::=} & \NTerm{MulExpr} ((\texttt{+}|\texttt{-})\NTerm{MulExpr})*   \\
      \NTerm{MulExpr}  & \mathtt{::=} & \NTerm{PrimExpr} ((\texttt{*}|\texttt{/})\NTerm{PrimExpr})* \\
      \NTerm{PrimExpr} & \mathtt{::=} & \Term{(}\ \NTerm{AddExpr}\ \Term{)}\ |\ \Term{Literal}
    \end{array}\]

  $\Term{Literal}$は数のトークンであるとする。

  \greentext{\small $\NTerm{AddExpr}$と$\NTerm{MulExpr}$にはそれぞれ１つしか生成規則はなく，
    $\NTerm{PrimExpr}$については、$\First{(\Term{(}\ \NTerm{AddExpr}\ \Term{)}}=\{\Term{(}\}$，
    $\First(\Term{Literal})$は互いに素であることから$ELL(1)$文法である．}
