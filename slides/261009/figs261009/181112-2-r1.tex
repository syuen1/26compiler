\begin{frame}
 \frametitle{LR(1)項}
 \greentext{LR項を先読み文字で拡張}
 \begin{center}
  \begin{Def}
   \begin{trivlist}
    \item [(LR項)]
   生成規則 $A\rightarrow\beta_1\beta_2$ ($\beta_1\in(N\cup\Sigma)^\ast$,
   $\beta_2\in(N\cup\Sigma)^\ast\cup(N\cup\Sigma)^\ast\#)$)
   $\beta_1$と$\beta_2$の間に$\bullet$を挿入した
   $A\rightarrow\beta_1\bullet\beta_2$項

   \item [(LR(1)項)]
LR項と
   $a\in\Sigma\cup\{\#,\varepsilon\}$の組
   $[A\rightarrow\beta_1\bullet\beta_2,a]$\\
	 ただし、$a=\varepsilon$となるのは、$A$が開始記号の場合のみ。
   \end{trivlist}
  \end{Def}
  
  \vspace*{-1em}

  \begin{Def}
   最右導出 $S'\Rightarrow_{rm}^\ast\alpha A
   w\Rightarrow_{rm}\alpha\beta_1\beta_2 w$が存在し、$a=First_1(w)$のと
   き、$[A\rightarrow\beta_1\bullet\beta_2,a]$は、可延頭部
   $\alpha\beta_1$に対する正準LR(1)項\\
   $I_1(\alpha\beta_1)$\ :\ $\alpha\beta_1$に対する正準LR(1)項の集合
  \end{Def}

\begin{frame}
 \frametitle{正準LR(1)オートマトン$\rightarrow$PDT}
 スタックトップ$I_{top}=I(X_1\ldots X_n)$
 \begin{trivlist}
  \item [{\bf 還元(Reduce)}]
	$[A\rightarrow\alpha\bullet,a]\in I_{top}$のとき

	$\alpha'\alpha w\Rightarrow_{rm}\alpha'Aw$\\
	$A\rightarrow\alpha$の規則番号$i$

	$\alpha=X_{n-k+1}\ldots X_n$とする($\alpha$の長さが$k$)とき、ス
	タックトップから\\
	$X_{n-k+1}I_1(X_1\ldots X_{n-k+1})\ldots X_n I_1(X_1\ldots X_n)$
	を$AI_1(X_1\ldots X_{n-k}A)$でおきかえる=$r_i$動作
	\[
	 \begin{array}{ll}
	  & (I_1(\varepsilon)\ldots X_{n-k}I_1\underline{(X_1\ldots
	   X_{n-k})\ldots  X_nI_1(X_1\ldots X_n)},ax,\pi)\\
	  \vdash 
	  & (I_1(\varepsilon)\ldots X_{n-k}I_1(X_1\ldots X_{n-k})A
	   I_1(X_1\ldots X_{n-k}A),ax,\pi i)\\
	 \end{array}
	\]

	\greentext{LR(1)構文解析表が同様に構成できる。}

	$I_1(X_1\ldots X_{n-k})=I_l$とするとき、
	$\mathsf{Action}(l,a)=r_i$
	$\mathsf{Goto}(l,A)=m$であれば$I_m\in I_1(X_1\ldots X_{n-k}A)$
 \end{trivlist}
\end{frame}

\begin{frame}
 \frametitle{正準LR(1)オートマトン$\rightarrow$PDT}
 スタックトップ$I_{top}=I_1(X_1\ldots X_n)$
 \begin{trivlist}
  \item [{\bf シフト(Shift)}]
	$[A\rightarrow\beta\bullet a\gamma,b]\in I_{top}$のとき

	\[
	 \begin{array}{ll}
	  & (I_1(\varepsilon)\ldots X_nI_1(X_1\ldots X_n),ax,\pi)\\
	  \vdash 
	  & (I_1(\varepsilon)\ldots X_nI_1(X_1\ldots X_n)aI_1(X_1\ldots X_na),x,\pi)\\
	 \end{array}
	\]

	$I_1(X_1\ldots X_n)=I_l$ $I_1(X_1\ldots X_na)=I_j$のとき、
	$\mathsf{Action}(l,a)=s_j$
  \item [{\bf 受理(Accept)}]
	$[S'\rightarrow S\bullet\#,\varepsilon]\in I_{top}$ならば受理
	\[
	 (I_1(\varepsilon)SI_1(S),\#,\pi)\vdash(\mbox{受理},\#,\pi)
	\]
	受理動作=$acc$
  \item [{\bf 誤り(Error)}]
	上記いずれにもあてはまらないとき
 \end{trivlist}
\end{frame}
