\begin{frame}
  \frametitle{上昇型構文解析}
  $A\rightarrow\alpha$の右辺へのマッチングによっ
  て$A$に還元する

  出力：最右導出の逆順\\
  \greentext{入力(左から)に従って下側から還元}

  \[
    \begin{array}{cl}
      w = w_1w_2 \Leftarrow N_1w_2                    & \mbox{$N_1\rightarrow w_1$
      によって還元}                                                                      \\
      N_1w_2 = N_1w_{21}w_{22}\Leftarrow N_1N_2w_{22} &                            \\
    \end{array}
  \]

  \greentext{\small 右側の還元が最後になる=最右導出の逆順}

  \redtext{難しい点：$\alpha(i)$に$\Sigma$と$N$が混在}

  還元するときのルールの選択がキーポイント
\end{frame}

    \item[(2)] スタックトップが$I_i$で、
          $[A\rightarrow X_1\cdots X_n\bullet]\in I_i$、先読み文字
          $a\in Follow_1(A)$のとき、$X_1I_{j_1}X_2I_2I_{j_2}\cdots
            X_nI_i$をpopする。このときスタックトップが
          $I_k$とすると$AI_\ell$をpushする。(ここで、$j$はGoto表
          の$(I_i,A)$に書かれている$I_j$である。)
  \end{list}
\end{frame}
