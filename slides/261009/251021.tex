\begin{frame}
  \frametitle{LR構文解析}
  PDTを用いた上昇型構文解析：最右導出の逆順を求める\\
  \greentext{先の例：(無あいまいであれば)13665176の順で生成規
    則を適用する最右導出によって解析木を構成可能}
  \[
    \begin{array}{rl}
      A\rightarrow & \alpha_1\cdots\alpha_n \\\cline{2-2}
                   & \mbox{左からマッチングをとる}
    \end{array}
  \]

  \begin{list}{}{}
    \item $\alpha_i\in\Sigma$のとき、入力文字を読みこむ\\
          \redtext{Shift動作}
    \item $\alpha_i\in N$のとき、一旦$\alpha$をスタックにつんで
          再帰的にマッチング動作を繰りかえす
    \item 最終的に$\alpha_i$がマッチすることがわかれば還元\\
          \redtext{Reduce動作}
  \end{list}
  この規則の最後までいけば全体を$A$に還元する

\begin{frame}

  {\small
  以下の手順に沿って構成される遷移系$\Theta=(\mathcal{S},\mathcal{E})$を{\bf 正準LR(0)遷移系}とよぶ．
  $\mathcal{S}$は$\Theta$の状態，$\mathcal{E}$は$\Theta$の辺である．
  \begin{list}{{\rm\bf ステップ\arabic{coffee}}:}{\usecounter{coffee}}
    \item $I_0=\mathsf{Goto}([S'\rightarrow S\texttt{\#}\,],\varepsilon)$とする．
    \item $\Theta\leftarrow(\{I_0\},\varnothing)$;$T\leftarrow\{I_0\}$
    \item\label{loopLR0} $T=\varnothing$ならば終了する．$\Theta$に正準LR(0)遷移系が得られる．
    \item $I\in T$を選び，$T$から削除する．
    \item すべての$M\in\mathsf{act}(I)$に対して，$I'=\{\theta'\,|\,\theta\in I,\theta
            \xrightarrow{\varepsilon}^\ast\xrightarrow{M}\xrightarrow{\varepsilon}^\ast\theta'\}$
          を構成する．
          \begin{list}{}{}
            \item $I'\notin\mathcal{S}$ならば，$T$に$I'$を
                  $\mathcal{S}$に$I'$を追加し，$\mathcal{E}$に$I\xrightarrow{M}I'$を追加する．
            \item $I'\in\mathcal{S}$ならば，$\mathcal{E}$に$I\xrightarrow{M}I'$を追加する．
          \end{list}
    \item ステップ\ref{loopLR0}へもどる．
  \end{list}
  }
