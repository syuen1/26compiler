  \begin{CustomDef}{7.1}{定義合流条件}\label{def:definitionConfluence}
    変数$x$に対する$\varphi$関数が頂点$v$に挿入$\Leftrightarrow$ 以下に示す条件(1)$\sim$(4)を満たす
    コントロールグラフ上の路$p:v_0\rightarrow v_1\rightarrow\cdots\rightarrow v_k$,
    $q:u_0\rightarrow u_1\rightarrow\cdots\rightarrow u_\ell$が存在．
    \begin{list}{(\arabic{coffee})}{\usecounter{coffee}}
        \item 路の長さが1以上である．($k>0$,$\ell>0$);
        \item $v_1$と$u_1$の両方に$x$の定義が存在する．;
        \item 路の終点$v$が一致している．($v_k=v_\ell=v$);
        \item 路の途中で交わることがない．($\{v_1,\cdots,v_{k-1}\}\cap\{u_1,\cdots,v_{\ell-1}\}=\varnothing$)
    \end{list}
  \end{CustomDef}

  パス$p$上の$x$の定義=$x_1$,$q$上の$x$の定義=$x_2$
  定義合流条件の$v$に、$x_3=\phi(x_1,x_2)$

\begin{frame}
  \frametitle{支配関係}

  支配関係：開始頂点からのパスに必ず含まれるとき、「支配」される。

  \only<1>{
    % \begin{center}
    %   \includegraphics[width=.7\textwidth]{def6_23.pdf}
    % \end{center}

    \begin{CustomDef}{7.2}{支配関係}
    フローグラフを$(V,E,v_{init})$とする．
    頂点$v,u$に対して，$v_{init}$から$u$へのすべての路に
    おいて$v$が出現するとき，$v$は$u$の{\bf 支配頂点(dominator)}であると呼び，$u\domll v$と書く．
    $u\not= v$かつ$u\domll v$であるとき，$u\ll v$と書く．
    $u$の支配頂点の集合$\domv{u}=\{v\;|\;u\domll v\}$で定義する．
    \end{CustomDef}

    $u\underline{\gg}v$:$u$が$v$を支配:支配関係の表現=支配木 %(図6.7)
  }
  \only<2>{
    どの頂点$v$についても$v_{init}\in\domv{v}$かつ$v\in\domv{v}$である．$\domll$は半順序である．
\begin{itemize}
    \item (反射律) $v\domll v$
    \item (反対称律) $v\domll u$かつ$u\domll v$ならば$u=v$
    \item (推移律) $v\domll u$かつ$u\domll w$ならば$v\domll w$
\end{itemize}

\vspace*{10pt}

$u\not= v_{init}$のとき，$\domv{u}\backslash\{u\}$には$\domll$の意味で最小の頂点が一つ存在する．
  }

\begin{frame}
  \frametitle{静的単一代入を用いた最適化}

  データフロー解析のもうひとつの手法:\\
  静的単一代入(Static Single Assignment:SSA)\\
  \redtext{各変数の定義がプログラム中で唯一}

  二度目の代入が発生すると変数名を変更\\
  % \quad\greentext{変数を見ればUdチェーンがすぐに判明する。}\\
  \quad\greentext{値が同じの共通部分式が判別できる=式として一致}

  %  変換例：例6.20、6.21\\
  定義ごとに変数に添字をつける。合流する場合は、$\varphi$関数で表現

  いかにして、SSA形式を得るか？

  \vspace*{5pt}

  \begin{list}{}{}
    \item [(1)] $\varphi$関数の挿入
    \item [(2)] 変数の添字の決定
  \end{list}
  \par\vspace*{3pt}\par
  \qquad\redtext{SSA形式において最適化}
  \par\vspace*{3pt}\par
  \begin{list}{}{}
    \item [(3)] SSAから逆変換する。($\varphi$関数の消去)
  \end{list}

  \greentext{近年のコンパイラでよく用いられる手法}

\end{frame}

  \frametitle{$\varphi$関数の挿入}

%  フローグラフ：$(V,E,v_{init})$

  $\texttt{x=}\varphi(\texttt{x},\cdots,\texttt{x})$の挿入：

  \vspace*{10pt}

  \begin{quote}
    \begin{trivlist}
    \item[$\cdot$] 変数\verb|x|に対して定義7.1の条件を満たす$v_0$, $u_0$, $v$がフローグラフに存在し，頂点$v$の基本ブロックに$\varphi$関数が含まれていないならば，$v$に\verb|x=|$\varphi$\verb|(x,...,x)|を挿入する．\\[10pt]
    \item[$\cdot$] 定義合流条件を満たす$v_0$, $u_0$, $v$ がなくなるまで，上記を繰り返す．
        \end{trivlist}
  \end{quote}
\end{frame}
