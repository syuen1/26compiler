class CalcParser(Parser):
    # 利用するトークン定義を Lexer から受け取る
    tokens = CalcLexer.tokens

    # 演算子の優先順位と結合性（下ほど優先順位が高い）
    precedence = (
        ('left', '+', '-'),
        ('left', '*', '/'),
        ('right', 'UMINUS'),  # 単項マイナス
    )

    # 文法規則: expr の計算

    @_('expr "+" expr')
    def expr(self, p):
        return p.expr0 + p.expr1

    @_('expr "-" expr')
    def expr(self, p):
        return p.expr0 - p.expr1

    @_('expr "*" expr')
    def expr(self, p):
        return p.expr0 * p.expr1

    @_('expr "/" expr')
    def expr(self, p):
        if p.expr1 == 0:
            print("Error: division by zero")
            return 0
        return p.expr0 / p.expr1\end{Verbatim}
