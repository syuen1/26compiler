\begin{frame}[fragile]
  \frametitle{LL(1)とLALR(1)}

  LL(1)ではあるがLALR(1)でない文法
  \[\left\{
    \begin{array}{clcl}
      1: & S\rightarrow \verb|(|X & 5: & X\rightarrow F\verb|]| \\
      2: & S\rightarrow E\verb|]| & 6: & E \rightarrow A        \\
      3: & S\rightarrow F\verb|)| & 7: & F\rightarrow A         \\
      4: & X\rightarrow E\verb|)| & 8: & A\rightarrow
    \end{array}
    \right\}
  \]

  LALR(1)であるがLL(1)でない文法
  \[
    \left\{
    \begin{array}{clcl}
      1: & S\rightarrow A\verb|a|         & 4: & S\rightarrow \verb|b|B\verb|a| \\
      2: & S\rightarrow \verb|b|A\verb|c| & 5: & A\rightarrow\verb|d|           \\
      3: & S\rightarrow B\verb|c|         & 6: & B\rightarrow\verb|d|
    \end{array}\right\}
  \]
\end{frame}

\begin{frame}
  \frametitle{演算子順位構文表}
  \[
    \begin{array}{c||c|c|c|c|c|c}
             & i,n & +,- & \ast,/ & (   & )   & \#  \\\hline\hline
      i,n    &     & \gg & \gg    &     & \gg & \gg \\\hline
      +,-    & \ll & \gg & \ll    & \ll & \gg & \gg \\\hline
      \ast,/ & \ll & \gg & \gg    & \ll & \gg & \gg \\\hline
      (      & \ll & \ll & \ll    & \ll & ==  &     \\\hline
      )      &     & \gg & \gg    &     & \gg & \gg \\\hline
      \#     & \ll & \ll & \ll    & \ll & \ll & acc \\\hline
    \end{array}
  \]

  \vspace*{1zw}

  ハンドラ：$\ll$と$\gg$で囲まれた部分

  \greentext{還元して非終端記号に置き換える}
\end{frame}

\subsection{エラー解析}

\begin{frame}
  \frametitle{構文エラー表示と誤り回復}

  \begin{list}{}{}
    \item $w\notin L(G)$の場合は実行する必要はない．

    \item 構文解析時に構文解析表に空欄に至った場合エラーが発生．

    \item ユーザーに対してエラーの原因を提示して，修正を促す．

          \begin{itemize}
            \item 正しい構文解析が可能な修正を最小限行う．\\
                  文法が限定される．修正が限定的でないと難しい．
                  （スコアで提示するなどの方法はコストが大きいため，
                  あまり用いられない)
            \item パニックモードに入り，特定の分離記号(デリミター)
                  まで読みとばす．\texttt{\}}や\texttt{;}など．
          \end{itemize}

    \item 回復することによって複数のエラーを指摘することが可能になる．２つめ
          以降は回復に依存

  \end{list}
\end{frame}
