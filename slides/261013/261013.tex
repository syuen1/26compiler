\begin{frame}
  \frametitle{LR(1)項}
  \greentext{LR項を先読み文字で拡張}
  {\small\begin{itembox}[l]{定義(LR項)}
      \begin{itemize}
        \item 生成規則 $A\rightarrow\beta_1\beta_2$ ($\beta_1\in(N\cup\Sigma)^\ast$,
              $\beta_2\in(N\cup\Sigma)^\ast\cup(N\cup\Sigma)^\ast\#)$)
              $\beta_1$と$\beta_2$の間に$\bullet$を挿入した
              $A\rightarrow\beta_1\bullet\beta_2$項

        \item
              LR項と$a\in\Sigma\cup\{\#,\varepsilon\}$の組
              $[A\rightarrow\beta_1\bullet\beta_2,a]$\\
              ただし、$a=\varepsilon$となるのは、$A$が開始記号の場合のみ。
      \end{itemize}
    \end{itembox}

    \vspace*{10pt}

    \begin{itembox}[l]{正準LR(1)項}
      \begin{itemize}
        \item 最右導出 $S'\Rightarrow_{rm}^\ast\alpha A
                w\Rightarrow_{rm}\alpha\beta_1\beta_2 w$が存在し、$a=First_1(w)$のと
              き、$[A\rightarrow\beta_1\bullet\beta_2,a]$は、可延頭部
              $\alpha\beta_1$に対する正準LR(1)項
        \item
              $I_1(\alpha\beta_1)$\ :\ $\alpha\beta_1$に対する正準LR(1)項の集合
      \end{itemize}
    \end{itembox}}

\begin{frame}
  \frametitle{正準LR(1)遷移系$\rightarrow$PDT}
  スタックトップ$I_{top}=I_n=\mathsf{Goto}(I_1,X_1\ldots X_n)$
  \begin{trivlist}
    \item [{\bf 還元(Reduce)}]
    $[A\rightarrow\alpha\bullet,a]\in I_{top}$のとき

    $\alpha'\alpha w\Rightarrow_{rm}\alpha'Aw$\\
    $A\rightarrow\alpha$の規則番号$i$

    $\alpha=X_{n-k+1}\ldots X_n$とする($\alpha$の長さが$k$)とき、ス
    タックトップから\\
    $X_{n-k+1}X_{n-k+1}\ldots X_n I_n$
    を$AI_n'$でおきかえる．$I_n'=\mathsf{Goto}(I_{n-k},A)$
    \[
      \begin{array}{ll}
         & (I_1\ldots X_{n-k}I_{n-k}
        \underline{X_{n-k+1}\ldots  X_nI_n},{\color{red}a}x,\pi) \\
        \vdash
         & (I_1\ldots X_{n-k}I_{n-k}A
        I'_n,{\color{red}a}x,\pi i)                              \\
      \end{array}
    \]

    \greentext{LR(1)構文解析表が同様に構成できる。}

    $\mathsf{Goto}(I_1,X_1\ldots X_{n-k})=I_{n-k}$とするとき、
    $\mathsf{Action}(l_{n-k},{\color{red}a})=r_i$
    $\mathsf{Goto}(l_{n-k},A)=I'$であれば$I'=\mathsf{Goto}(I_1,X_1\ldots X_{n-k}A)$
  \end{trivlist}
\end{frame}

\begin{frame}
  \frametitle{正準LR(1)遷移系$\rightarrow$PDT}
  スタックトップ$I_{top}=I_n=\mathsf{Goto}(I_1,X_1\ldots X_n)$
  \begin{trivlist}
    \item [{\bf シフト(Shift)}]
    $[A\rightarrow\beta\bullet a\gamma,b]\in I_{top}$のとき

    \[
      \begin{array}{ll}
         & (I_1\ldots X_nI_n,ax,\pi)     \\
        \vdash
         & (I_1\ldots X_nI_naI'_n,x,\pi) \\
      \end{array}
    \]

    $\mathsf{Goto}(I_1,X_1\ldots X_n)=I_n$ $\mathsf{Goto}(X_1\ldots X_na)=I_j$のとき、
    $\mathsf{Action}(l_n,a)=s_j$
    \item [{\bf 受理(Accept)}]
    $[S'\rightarrow S\bullet\#,\varepsilon]\in I_{top}$ならば受理
    \[
      (I_1SI_2,\#,\pi)\vdash(acc,\#,\pi) \mbox{ここで}I_2=\mathsf{Goto}(I_1,S)
    \]
    \item [{\bf 誤り(Error)}]
    上記いずれにもあてはまらないとき
  \end{trivlist}
\end{frame}

\subsection{エラー解析}

\begin{frame}
  \frametitle{構文エラー表示と誤り回復}

  \begin{list}{}{}
    \item $w\notin L(G)$の場合は実行する必要はない．

    \item 構文解析時に構文解析表に空欄に至った場合エラーが発生．

    \item ユーザーに対してエラーの原因を提示して，修正を促す．

          \begin{itemize}
            \item 正しい構文解析が可能な修正を最小限行う．\\
                  文法が限定される．修正が限定的でないと難しい．
                  （スコアで提示するなどの方法はコストが大きいため，
                  あまり用いられない)
            \item パニックモードに入り，特定の分離記号(デリミター)
                  まで読みとばす．\texttt{\}}や\texttt{;}など．
          \end{itemize}

    \item 回復することによって複数のエラーを指摘することが可能になる．２つめ
          以降は回復に依存

  \end{list}
\end{frame}
