  \begin{Verbatim}[numbers=left]
int x,y,z;
int exp(int x,int y) {
  int rv;
  if (y==0) {rv=1;}
  else if (y<0) {rv=exp(x,y+1)/x;}
  else {rv=x*exp(x,y-1);}
  return rv;
}

main() {
  x=2; z=7; y=exp(x,z);
  printf("%d\n",y);
}
  \end{Verbatim}
